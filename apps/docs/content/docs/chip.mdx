import { Chip } from "@/registry/new-york/chip/chip";
import { ChipSelectableDemo, ChipRemovableDemo } from "@/components/demos/chip-demo";

The HBD chip is a compact, token-backed pill used for tags, filters, and
attributes. It renders as a static `<span>` by default, a toggle `<button>`
when `selectable`, and gains an inline remove (✕) button when `removable`.

## Removable

Add `removable` for an inline remove button. `onRemove({ label })` fires when
it is activated — drop the chip from your own state to dismiss it.

<Tabs items={['Preview', 'Code']}>
  <Tab value="Preview">
    <Preview>
      <ChipRemovableDemo />
    </Preview>
  </Tab>
  <Tab value="Code">
    ```tsx
    const [tags, setTags] = React.useState(['Compass', 'Astrolabe', 'Sextant']);

    {tags.map((label) => (
      <Chip
        key={label}
        variant="primary"
        removable
        onRemove={() => setTags((prev) => prev.filter((t) => t !== label))}
      >
        {label}
      </Chip>
    ))}
    ```

  </Tab>
</Tabs>

## Selectable

With `selectable`, the chip renders as a toggle `<button>` exposing
`aria-pressed`. It is controlled via `selected`; `onToggle({ selected, label })`
signals the next state.

<Tabs items={['Preview', 'Code']}>
  <Tab value="Preview">
    <Preview>
      <ChipSelectableDemo />
    </Preview>
  </Tab>
  <Tab value="Code">
    ```tsx
    const [selected, setSelected] = React.useState<string[]>(['Dragons']);

    <Chip
      selectable
      selected={selected.includes('Dragons')}
      onToggle={({ selected: next, label }) =>
        setSelected((prev) =>
          next ? [...prev, label] : prev.filter((l) => l !== label),
        )
      }
    >
      Dragons
    </Chip>
    ```

  </Tab>
</Tabs>
