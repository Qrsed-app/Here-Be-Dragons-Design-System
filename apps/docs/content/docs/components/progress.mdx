    ```tsx
    "use client";

    import * as React from "react";

    import { Progress } from "@/components/ui/progress";

    export function ProgressDemo() {
      const [progress, setProgress] = React.useState(13);

      React.useEffect(() => {
        const timer = setTimeout(() => setProgress(66), 500);
        return () => clearTimeout(timer);
      }, []);

      return <Progress value={progress} className="w-[60%]" />;
    }
    ```

## Installation

```bash
npx shadcn@latest add @hbd/progress
```

The `@hbd` namespace needs a one-time `registries` entry in `components.json` — see
[Installation](/docs/installation). Or add it by URL, no configuration needed:

```bash
npx shadcn@latest add http://localhost:3001/r/progress.json
```

## Usage

```tsx
import { Progress } from "@/components/ui/progress";
```

```tsx
<Progress value={33} />
```

### Label and value

Compose a `Field` with a `FieldLabel` and a readout above the bar, and point the bar at the label with
`aria-labelledby`.

## API Reference

<TypeTable
  type={{
    value: {
      type: "number | null",
      description: "Progress, from `0` to `max`.",
    },
    max: {
      type: "number",
      default: "100",
      description: "The value that counts as complete.",
    },
    variant: {
      type: '"default" | "success" | "warning" | "error"',
      default: '"default"',
      description: "Here Be Dragons addition. Fill colour.",
    },
    size: {
      type: '"sm" | "default" | "lg"',
      default: '"default"',
      description: "Here Be Dragons addition. Track height: 4, 8 or 12px.",
    },
    striped: {
      type: "boolean",
      default: "false",
      description:
        "Here Be Dragons addition. Animated diagonal stripes on the fill. Ignored when `indeterminate`.",
    },
    indeterminate: {
      type: "boolean",
      default: "false",
      description:
        "Here Be Dragons addition. Slides a chip along the track and reports an indeterminate state.",
    },
    segments: {
      type: "number",
      description:
        "Here Be Dragons addition. Renders this many cells instead of a continuous fill.",
    },
  }}
/>

Every other prop is forwarded to the Radix Progress root. See the
[Radix Progress documentation](https://www.radix-ui.com/primitives/docs/components/progress).
