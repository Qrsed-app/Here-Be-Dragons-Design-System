## Installation

```bash
npx shadcn@latest add @hbd/checkbox
```

The `@hbd` namespace needs a one-time `registries` entry in `components.json` — see
[Installation](/docs/installation). Or add it by URL, no configuration needed:

```bash
npx shadcn@latest add http://localhost:3001/r/checkbox.json
```

## Usage

```tsx
import { Checkbox } from "@/components/ui/checkbox";
```

```tsx
<Checkbox />
```

In a horizontal `Field`, `items-start gap-2 py-3` keeps the box level with the first line of the label
and the tap area at least 44px tall; the body-size label matches the other choice controls. The tick and
dash scale with the surrounding font size.

### Select all

The parent shows `"indeterminate"` while only some children are checked.

<Tabs items={["Preview", "Code"]}>
  <Tab value="Preview">
    <Preview>
      <CheckboxSelectAllDemo />
    </Preview>
  </Tab>
  <Tab value="Code">
    ```tsx
    "use client";

    import * as React from "react";

    import { Checkbox } from "@/components/ui/checkbox";
    import { Field, FieldLabel } from "@/components/ui/field";

    const instruments = ["Compass", "Astrolabe", "Sextant"];

    export function CheckboxSelectAllDemo() {
      const [packed, setPacked] = React.useState<string[]>(["Compass"]);
      const all = packed.length === instruments.length;

      return (
        <div className="flex flex-col gap-7">
          <Field orientation="horizontal" className="items-start gap-2 py-3">
            <Checkbox
              id="pack-all"
              checked={all ? true : packed.length > 0 ? "indeterminate" : false}
              onCheckedChange={(checked) => setPacked(checked === true ? [...instruments] : [])}
            />
            <FieldLabel htmlFor="pack-all" className="text-[1.0625rem] font-normal">
              Pack all instruments
            </FieldLabel>
          </Field>
          <div className="flex flex-col gap-5 ps-6">
            {instruments.map((item) => (
              <Field key={item} orientation="horizontal" className="items-start gap-2 py-3">
                <Checkbox
                  id={`pack-${item}`}
                  checked={packed.includes(item)}
                  onCheckedChange={(checked) =>
                    setPacked((prev) => (checked === true ? [...prev, item] : prev.filter((i) => i !== item)))
                  }
                />
                <FieldLabel htmlFor={`pack-${item}`} className="text-[1.0625rem] font-normal">
                  {item}
                </FieldLabel>
              </Field>
            ))}
          </div>
        </div>
      );
    }
    ```

  </Tab>
</Tabs>

## API Reference

`Checkbox` adds no props of its own. `checked` (`boolean | "indeterminate"`), `defaultChecked`,
`onCheckedChange`, `disabled`, `required`, `name`, `value` and every other prop go to the Radix
Checkbox root. `aria-invalid` shows the red border. See the
[Radix Checkbox documentation](https://www.radix-ui.com/primitives/docs/components/checkbox).
