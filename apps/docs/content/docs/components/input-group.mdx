## Installation

```bash
npx shadcn@latest add @hbd/input-group
```

The `@hbd` namespace needs a one-time `registries` entry in `components.json` — see
[Installation](/docs/installation). Or add it by URL, no configuration needed:

```bash
npx shadcn@latest add http://localhost:3001/r/input-group.json
```

## Usage

```tsx
import {
  InputGroup,
  InputGroupAddon,
  InputGroupButton,
  InputGroupInput,
  InputGroupText,
  InputGroupTextarea,
} from "@/components/ui/input-group";
```

```tsx
<InputGroup>
  <InputGroupInput placeholder="Search..." />
  <InputGroupAddon>
    <SearchIcon />
  </InputGroupAddon>
  <InputGroupAddon align="inline-end">
    <InputGroupButton>Search</InputGroupButton>
  </InputGroupAddon>
</InputGroup>
```

The group owns the border, focus ring and error, success, disabled and read-only surfaces: set
`aria-invalid`, `data-valid`, `disabled` or `readOnly` on the control and the group follows.

### Password Toggle

The old built-in eye toggle, as an icon button in an inline-end addon.

<Tabs items={["Preview", "Code"]}>
  <Tab value="Preview">
    <Preview>
      <InputGroupPasswordDemo />
    </Preview>
  </Tab>
  <Tab value="Code">
    ```tsx
    "use client";

    import * as React from "react";
    import { EyeIcon, EyeOffIcon } from "lucide-react";
    import {
      InputGroup,
      InputGroupAddon,
      InputGroupButton,
      InputGroupInput,
    } from "@/components/ui/input-group";

    export function InputGroupPasswordDemo() {
      const [visible, setVisible] = React.useState(false);

      return (
        <InputGroup className="max-w-sm">
          <InputGroupInput
            type={visible ? "text" : "password"}
            placeholder="••••••••"
            aria-label="Password"
          />
          <InputGroupAddon align="inline-end">
            <InputGroupButton
              size="icon-xs"
              aria-label={visible ? "Hide password" : "Show password"}
              aria-pressed={visible}
              onClick={() => setVisible((v) => !v)}
            >
              {visible ? <EyeOffIcon /> : <EyeIcon />}
            </InputGroupButton>
          </InputGroupAddon>
        </InputGroup>
      );
    }
    ```

  </Tab>
</Tabs>
